            'all_routes': routes,
            'reroute_reason': 'Main Highway Bridge structural risk exceeded 80% threshold.'
        }
    @staticmethod
    def optimize_resource_allocation(area_demands, available_inventory):
        """
        Resource Optimization Solver distributing available inventory based on Priority Scores
        """
        allocations = []
        for area in area_demands:
            score = area.get('severityScore', 50)
            affected = area.get('affected', 100)
            if score >= 85:
                amb = 2
                teams = 1
                water = 500
                food = 300
            elif score >= 60:
                amb = 1
                teams = 0
                water = 300
                food = 400
            else:
                amb = 0
                teams = 0
                water = 100
                food = 200
            allocations.append({
                'area_id': area.get('id'),
                'area_name': area.get('name'),
                'priority_score': score,
                'allocated_resources': {
                    'ambulances': amb,
                    'rescue_teams': teams,
                    'water_liters': water,
                    'food_kits': food
                },
                'justification': f'Prioritized score {score}/100 due to high critical patient count & road hazard.'
            })
        return allocations
